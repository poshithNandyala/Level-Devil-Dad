const px = (t: number) => t * 16;

const AR = T.actRoutes as Record<string, (x: number) => Step[]>;
function atAct(id: string, act: string, then: (x: number) => Step[]): Step[] {
  const def = LEVELS.find((l) => l.id === id)!;
  const out: Step[] = [...S[id]];
  for (const p of (def as any).parts ?? []) {
    if (p.kind === act) return [...out, ...then(p.x0)];
    out.push(...AR[p.kind](p.x0));
  }
  throw new Error(id + ' no act ' + act);
}
function allActs(id: string): Step[] {
  const def = LEVELS.find((l) => l.id === id)!;
  const out: Step[] = [...S[id]];
  for (const p of (def as any).parts ?? []) out.push(...AR[p.kind](p.x0));
  return out;
}
const toEnd = (note: string): Step => ({ hold: 'R', until: (w: World) => w.state !== 'play', max: 3000, note });
function hopEdges(): Step {
  let jt = 0;
  return {
    policy: (w: World) => {
      const p = w.player;
      if (jt > 0) { jt--; return { move: 'R', jump: true }; }
      if (p.onGround && (!w.solidAt(p.x + p.w + 6, p.y + p.h + 2, 2, 2) || w.solidAt(p.x + p.w + 4, p.y + 3, 2, p.h - 8))) { jt = 11; return { move: 'R', jump: true }; }
      return { move: 'R', jump: false };
    },
    until: (w: World) => w.state !== 'play',
    max: 4000,
  } as Step;
}
const go = (m: 'L' | 'R' | '') => ({ move: m, jump: false });
const toward = (w: World, x: number, s = 2) => (w.player.cx < x - s ? 'R' : w.player.cx > x + s ? 'L' : '') as 'L' | 'R' | '';

  const out: string[] = [];
  for (const [id, what, mk] of PLAYS) {
    const def = LEVELS.find((l) => l.id === id)!;
    try {
      const { w } = play(def, mk());
      out.push(`${id} | ${what} | ${w.state}${w.state === 'dead' ? ' ' + w.deathCause : ''}@${w.tick} x${(w.player.x / 16).toFixed(1)}`);
    } catch (e) {
      out.push(`${id} | ${what} | ERROR ${(e as Error).message.slice(0, 160)}`);
    }
  }
  console.log('QAP\n' + out.join('\n'));
}, 600000);
