import { it } from 'vitest';
import { NAIVE, FIX, MISTAKES } from '../solutions';
import { causeLines } from '../../src/game/cities/dis/causes';
// QA scratch: the labels behind m0..mN in the sweep output, and every DIS cause line
it.runIf(!!process.env.QAL)('qa dis labels', () => {
  const out: string[] = [];
  for (let i = 1; i <= 10; i++) {
    const id = `DIS-${String(i).padStart(2, '0')}`;
    out.push(`${id} naive [${NAIVE[id]?.cause ?? '-'}] ${(NAIVE[id] as any)?.what ?? ''} | fix [${FIX[id]?.cause ?? '-'}] ${(FIX[id] as any)?.what ?? ''}`);
    (MISTAKES[id] ?? []).forEach((m, k) => out.push(`  m${k} [${m.cause}] ${m.what}`));
  }
  out.push('CAUSES ' + Object.keys(causeLines).length);
  for (const [k, v] of Object.entries(causeLines)) out.push(`  ${k}: ${v.join(' / ')}`);
  console.log('LABELS\n' + out.join('\n'));
});
