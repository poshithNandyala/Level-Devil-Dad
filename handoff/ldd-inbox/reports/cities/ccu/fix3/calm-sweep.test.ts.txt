import { it } from 'vitest';
import { LEVELS } from '../src/game/levels';
import { SOLUTIONS, NAIVE, FIX, MISTAKES } from './solutions';
import { World, type InputState } from '../src/game/world';
import { incoming, windOn, type Step } from './runner';
import { INFO } from './helpers';
import { encodeInput, replay } from '../src/game/replay';

let REC: number[] = [];
function lagged(id: string, steps: Step[], lag: number, idle: number, cpt: number[]) {
  const def = LEVELS.find((l) => l.id === id)!;
  const w = new World(def, INFO);
  def.build?.(w);
  const none: InputState = { left: false, right: false, jump: false, jumpPressed: false };
  // left and right go down `lag` ticks after the route presses them and come up as soon as it lets go;
  // a jump goes down `lag` ticks late and is held just as long as the route held it
  const keys = ['left', 'right'] as const;
  const want: Record<string, number> = { left: -1, right: -1 };
  const down: Record<string, boolean> = { left: false, right: false };
  // a short press (a nudge, a step off) still happens in full, just late
  const pressed: Record<string, number> = { left: -1, right: -1 };
  const until: Record<string, number> = { left: -1, right: -1 };
  const jumps: { at: number; len: number }[] = [];
  let routeJump = false;
  let lastJump = -99;
  let tick = 0;
  let prevApplied = false;
  const push = (inp: InputState) => {
    for (const k of keys) {
      if (!inp[k]) {
        // a direction pressed together with a jump is part of the jump: it's held its full length, late
        if (pressed[k] >= 0 && Math.abs(pressed[k] - lastJump) <= 2) until[k] = pressed[k] + lag + (tick - pressed[k]);
        pressed[k] = -1;
        want[k] = -1;
        down[k] = tick < until[k];
      } else {
        if (pressed[k] < 0) pressed[k] = tick;
        if (!down[k]) {
          if (want[k] < 0) want[k] = tick + lag;
          if (tick >= want[k]) down[k] = true;
        }
      }
    }
    if (down.left && down.right) down[inp.left ? 'right' : 'left'] = false;
    if (inp.jump && !routeJump) {
      jumps.push({ at: tick + lag, len: 1 });
      lastJump = tick;
    }
    else if (inp.jump && jumps.length) jumps[jumps.length - 1].len++;
    routeJump = inp.jump;
    while (jumps.length && jumps[0].at + jumps[0].len <= tick) jumps.shift();
    const jump = jumps.length > 0 && tick >= jumps[0].at;
    const out = { left: down.left, right: down.right, jump, jumpPressed: jump && !prevApplied };
    prevApplied = jump;
    tick++;
    const before = w.cp;
    REC.push(encodeInput(out));
    w.step(out);
    if (w.cp !== before) cpt.push(REC.length);
  };
  for (let t = 0; t < idle && w.state === 'play'; t++) push(none);
  let jumpLeft = 0;
  for (const s of steps) {
    let t = 0;
    const max = (s.max ?? 3000) + lag + 240;
    if (s.jump) jumpLeft = s.jump;
    while (w.state === 'play') {
      if (s.dodge && w.player.onGround && jumpLeft <= 0 && incoming(w, s.dodge)) jumpLeft = 12;
      const lean = s.brace ? windOn(w) : 0;
      const pol = s.policy ? s.policy(w) : null;
      const hold = pol ? pol.move : s.steer ? s.steer(w) : s.hold;
      push({
        left: hold === 'L' || (hold === 'FWD' && w.flipped) || lean > 0,
        right: hold === 'R' || (hold === 'FWD' && !w.flipped) || lean < 0,
        jump: pol ? pol.jump || jumpLeft > 0 : jumpLeft > 0,
        jumpPressed: false,
      });
      if (jumpLeft > 0) jumpLeft--;
      t++;
      if (s.ticks !== undefined && t >= s.ticks) break;
      if (s.until && s.until(w)) break;
      if (t >= max) return { w, won: false, stuck: true };
    }
    if (w.state !== 'play') break;
  }
  let guard = 0;
  while (w.state === 'play' && guard++ < 30) push(none);
  return { w, won: w.state === "won" };
}



// lag-captured calm sweep (the tester's way): route at each lag, capture where that run captures, then
// from each capture: stand idle 0 / 0.5 / 1 s first (the spawn idle), then idle 4 s, run 0.3 s, tap-jump
it('ccu calm sweep', () => {
  const out: string[] = [];
  const I = { left: false, right: false, jump: false, jumpPressed: false };
  const R = { ...I, right: true };
  const plays: [string, (i: number) => any][] = [
    ['idle4', () => I],
    ['run.3', (i) => (i < 36 ? R : I)],
    ['tap6', (i) => (i < 40 ? { ...R, jump: i < 6, jumpPressed: i === 0 } : I)],
    ['tap12', (i) => (i < 40 ? { ...R, jump: i < 12, jumpPressed: i === 0 } : I)],
  ];
  const SH: ((i: number, w: any) => any)[] = [];
  // the shared cpsafe plays too: run 0.3 s after `lag`, and a 6-tick tap-jump running on until he lands
  for (const lag of [0, 6, 12]) {
    let landed = false;
    SH.push((i: number, w: any) => (i >= lag && i < lag + 36 ? R : I));
    SH.push((i: number, w: any) => {
      if (i === 0) landed = false;
      if (i < lag || landed) return I;
      const k = i - lag;
      if (k > 6 && w.player.onGround) landed = true;
      if (landed || k > 240) return I;
      return { ...R, jump: k < 6, jumpPressed: k === 0 };
    });
  }
  SH.forEach((f, k) => plays.push([`shared${k}`, f as any]));
  const ids = (process.env.IDS ?? LEVELS.filter((l) => l.id.startsWith('CCU')).map((l) => l.id).join(',')).split(',');
  for (const id of ids) {
    const def = LEVELS.find((l) => l.id === id)!;
    let bad = 0;
    for (const lag of [0, 6, 11, 12]) for (const sidle of [0, 37, 120, 251]) {
      REC = [];
      const cpt: number[] = [];
      const r: any = lagged(id, SOLUTIONS[id], lag, sidle, cpt);
      if (!r.won) { out.push(`${id} L${lag} i${sidle} ROUTE LOST ${r.w.deathCause}`); bad++; continue; }
      const rec = REC.slice();
      cpt.forEach((t, k) => {
        for (const pre of [0]) for (const [name, f] of plays) {
          const w = replay(def, INFO, rec.slice(0, t));
          w.player.vx = 0;
          for (let i = 0; i < pre && w.state === 'play'; i++) w.step(I);
          for (let i = 0; i < 480 && w.state === 'play'; i++) w.step((f as any)(i, w));
          if (w.state === 'dead') { bad++; out.push(`${id} L${lag} i${sidle} cp${k} x${(w.checkpoints[k] / 16).toFixed(1)} idle${pre} ${name} DIES ${w.deathCause} +${((w.tick - t) / 120).toFixed(2)}s`); }
        }
      });
    }
    out.push(`${id} calm deaths ${bad}`);
  }
  console.log(out.map((l) => 'CALM ' + l).join('\n'));
}, 3600000);
