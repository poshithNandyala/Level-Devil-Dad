import { it } from 'vitest';
import { writeFileSync } from 'fs';
import { LEVELS } from '../../src/game/levels';
import { play } from '../runner';
import { NAIVE, FIX, MISTAKES } from '../solutions';
// QA scratch: record BBI NAIVE / FIX / MISTAKES plays to /tmp/qa-bbi-rec/<ID>__<kind>.json
it('qa bbi rec', () => {
  for (const def of LEVELS.filter((d) => d.id.startsWith('BBI'))) {
    const runs: [string, { steps: any[] }][] = [];
    if (NAIVE[def.id]) runs.push(['naive', NAIVE[def.id]]);
    if (FIX[def.id]) runs.push(['fix', FIX[def.id]]);
    (MISTAKES[def.id] ?? []).forEach((m, i) => runs.push([`m${i}`, m]));
    for (const [k, r] of runs) {
      (globalThis as any).__lddRec = [];
      const { w } = play(def, r.steps);
      const rec = (globalThis as any).__lddRec as number[];
      (globalThis as any).__lddRec = undefined;
      writeFileSync(`/tmp/qa-bbi-rec/${def.id}__${k}.json`, JSON.stringify({ rec, end: w.tick, state: w.state, cause: w.deathCause }));
      console.log(`${def.id} ${k} ${w.state} ${w.deathCause} t=${(w.tick / 120).toFixed(1)}s`);
    }
  }
});
