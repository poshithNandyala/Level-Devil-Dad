// scratch probe for the BBI length pass (not shipped): (a) identical windows per stage, (b) FIX vs NAIVE per act
import { it } from 'vitest';
import { LEVELS } from '../src/game/levels';
import { play } from './runner';
import { MISTAKES, SOLUTIONS } from './solutions';
import { CITY_TESTS } from './city-tests';
import { World, type InputState } from '../src/game/world';
import { incoming, windOn, type Step } from './runner';
import { INFO } from './helpers';

/** acts declared as a stage's one momentum beat: the same hazard and answer once more, faster */
const MOMENTUM = new Set(['bbi07-shrinkhala']);

it('BBI probes', () => {
  process.env.LDD_EVERY = '1';
  const out: string[] = [];
  let badA = 0;
  let badB = 0;
  for (const def of LEVELS.filter((l) => l.id.startsWith('BBI-') && (!process.env.ONLY || l.id === process.env.ONLY))) {
    const parts = (def as any).parts as { kind: string; x0: number }[] | undefined;
    if (!parts) continue;
    const sig: Record<string, string[]> = {};
    for (const p of parts) {
      const ms = (MISTAKES[def.id] ?? []).filter((m) => m.what.startsWith(p.kind + ':'));
      const res = ms.map((m) => {
        let entry = -1;
        (globalThis as any).__lddWatch = (w: any) => {
          if (entry < 0 && w.player.x >= p.x0 * 16) entry = w.tick;
          return '';
        };
        const { w } = play(def, m.steps, undefined, true);
        (globalThis as any).__lddWatch = undefined;
        return { what: m.what.slice(p.kind.length + 2), cause: w.deathCause ?? w.state, t: w.tick - entry };
      });
      const n = res[0];
      const f = res.length > 1 ? res[res.length - 1] : undefined;
      let b = 'NO DEATHS';
      if (n && !f) b = 'NO FIX';
      if (n && f) b = n.cause !== f.cause && n.t !== f.t ? 'ok' : 'SAME';
      if (b !== 'ok') badB++;
      out.push(`(b) ${def.id} ${p.kind.padEnd(18)} naive=${n ? `${n.cause}@${n.t}` : '-'} fix=${f ? `${f.cause}@${f.t}` : '-'} -> ${b}`);
      const key = n ? `${n.cause}|${f ? `${f.cause} (${f.what})` : '?'}` : `no per-act deaths: ${p.kind}`;
      (sig[key] ??= []).push(p.kind);
    }
    for (const [k, ks] of Object.entries(sig)) {
      const r = ks.length - ks.filter((k2) => MOMENTUM.has(k2)).length <= 1 ? 'ok' : ks.length <= 2 && !ks.some((k2) => MOMENTUM.has(k2)) ? 'ok (one momentum repeat)' : 'REPEAT';
      if (r === 'REPEAT') badA++;
      out.push(`(a) ${def.id} window ${k.padEnd(28)} x${ks.length} [${ks.join(', ')}] -> ${r}`);
    }
  }
  out.push(`probe (a) repeats beyond one momentum beat: ${badA}; probe (b) acts failing: ${badB}`);
  console.log(out.join('\n'));
}, 600000);

it('BBI route times', () => {
  const out: string[] = [];
  for (const def of LEVELS.filter((l) => l.id.startsWith('BBI-'))) {
    const { w } = play(def, SOLUTIONS[def.id]);
    out.push(`RT ${def.id} ${w.state} ${(w.tick / 120).toFixed(1)} s`);
  }
  console.log(out.join('\n'));
}, 600000);

function laggedTrace(id: string, steps: Step[], lag: number, idle: number, trace: { x: number; cp: number }[]) {
  const def = LEVELS.find((l) => l.id === id)!;
  const w = new World(def, INFO);
  def.build?.(w);
  const none: InputState = { left: false, right: false, jump: false, jumpPressed: false };
  // left and right go down `lag` ticks after the route presses them and come up as soon as it lets go;
  // a jump goes down `lag` ticks late and is held just as long as the route held it
  const keys = ['left', 'right'] as const;
  const want: Record<string, number> = { left: -1, right: -1 };
  const down: Record<string, boolean> = { left: false, right: false };
  // a short press (a nudge, a step off) still happens in full, just late
  const pressed: Record<string, number> = { left: -1, right: -1 };
  const until: Record<string, number> = { left: -1, right: -1 };
  const jumps: { at: number; len: number }[] = [];
  let routeJump = false;
  let lastJump = -99;
  let tick = 0;
  let prevApplied = false;
  const push = (inp: InputState) => {
    for (const k of keys) {
      if (!inp[k]) {
        // a direction pressed together with a jump is part of the jump: it's held its full length, late
        if (pressed[k] >= 0 && Math.abs(pressed[k] - lastJump) <= 2) until[k] = pressed[k] + lag + (tick - pressed[k]);
        pressed[k] = -1;
        want[k] = -1;
        down[k] = tick < until[k];
      } else {
        if (pressed[k] < 0) pressed[k] = tick;
        if (!down[k]) {
          if (want[k] < 0) want[k] = tick + lag;
          if (tick >= want[k]) down[k] = true;
        }
      }
    }
    if (down.left && down.right) down[inp.left ? 'right' : 'left'] = false;
    if (inp.jump && !routeJump) {
      jumps.push({ at: tick + lag, len: 1 });
      lastJump = tick;
    }
    else if (inp.jump && jumps.length) jumps[jumps.length - 1].len++;
    routeJump = inp.jump;
    while (jumps.length && jumps[0].at + jumps[0].len <= tick) jumps.shift();
    const jump = jumps.length > 0 && tick >= jumps[0].at;
    const out = { left: down.left, right: down.right, jump, jumpPressed: jump && !prevApplied };
    prevApplied = jump;
    tick++;
    w.step(out);
    trace.push({ x: w.player.x, cp: w.cp });
    if ((globalThis as any).__dbg) (globalThis as any).__dbg(w, out);
  };
  for (let t = 0; t < idle && w.state === 'play'; t++) push(none);
  let jumpLeft = 0;
  for (const s of steps) {
    let t = 0;
    const max = (s.max ?? 3000) + lag + 240;
    if (s.jump) jumpLeft = s.jump;
    while (w.state === 'play') {
      if (s.dodge && w.player.onGround && jumpLeft <= 0 && incoming(w, s.dodge)) jumpLeft = 12;
      const lean = s.brace ? windOn(w) : 0;
      const pol = s.policy ? s.policy(w) : null;
      const hold = pol ? pol.move : s.steer ? s.steer(w) : s.hold;
      push({
        left: hold === 'L' || (hold === 'FWD' && w.flipped) || lean > 0,
        right: hold === 'R' || (hold === 'FWD' && !w.flipped) || lean < 0,
        jump: pol ? pol.jump || jumpLeft > 0 : jumpLeft > 0,
        jumpPressed: false,
      });
      if (jumpLeft > 0) jumpLeft--;
      t++;
      if (s.ticks !== undefined && t >= s.ticks) break;
      if (s.until && s.until(w)) break;
      if (t >= max) return { won: false, cp: w.cp, cps: w.checkpoints.length, why: `stuck: ${s.note ?? ''}` };
    }
    if (w.state !== 'play') break;
  }
  let guard = 0;
  while (w.state === 'play' && guard++ < 30) push(none);
  return { won: w.state === "won", cp: w.cp, cps: w.checkpoints.length, why: w.state === "dead" ? `died: ${w.deathCause} @${Math.floor(w.player.x / 16)}` : w.state };
}


it('BBI replay to each trap from its checkpoint', () => {
  const out: string[] = [];
  let worst = 0;
  for (const def of LEVELS.filter((l) => l.id.startsWith('BBI-') && (!process.env.ONLY || l.id === process.env.ONLY))) {
    const cps = (() => { const { w } = play(def, []); return w.checkpoints.slice(); })();
    // where each scripted death happens (the trap)
    const traps = (MISTAKES[def.id] ?? []).map((m) => { const { w } = play(def, m.steps); return { what: m.what, x: w.player.x }; });
    for (const lag of [0, 6, 12]) {
      const tr: { x: number; cp: number }[] = [];
      laggedTrace(def.id, SOLUTIONS[def.id], lag, 0, tr);
      let maxR = 0;
      let at = '';
      for (const t of traps) {
        const reach = tr.findIndex((s) => s.x >= t.x - 8);
        if (reach < 0) continue;
        const c = cps.filter((x) => x <= t.x).pop();
        const from = c === undefined ? 0 : tr.findIndex((s) => s.x + 5 >= c);
        const r = (reach - Math.max(0, from)) / 120;
        if (r > maxR) { maxR = r; at = t.what; }
      }
      worst = Math.max(worst, maxR);
      if (lag === 0) {
        // the longest stretch of the route between two calm spots (checkpoint passes, then the door)
        let last = 0;
        let gap = 0;
        for (let i = 1; i < tr.length; i++) if (tr[i].cp !== tr[i - 1].cp) { gap = Math.max(gap, (i - last) / 120); last = i; }
        gap = Math.max(gap, (tr.length - last) / 120);
        out.push(`stretch ${def.id}: longest ${gap.toFixed(1)} s between calm spots`);
      }
      out.push(`replay ${def.id} lag ${String(lag).padStart(2)}: longest ${maxR.toFixed(1)} s (${at})${maxR > 8 ? '  OVER 8 s' : ''}`);
    }
  }
  out.push(`replay worst: ${worst.toFixed(1)} s`);
  console.log(out.join('\n'));
}, 900000);


it('dbg', () => {
  if (!process.env.DBG) return;
  const [id, a, b, lag] = process.env.DBG.split(',');
  const lines: string[] = [];
  (globalThis as any).__dbg = (w: any, o: any) => { if (w.player.x > +a * 16 && w.player.x < +b * 16 && w.tick % 4 === 0) lines.push(`t${w.tick} x=${(w.player.x / 16).toFixed(2)} y=${w.player.y} g=${w.player.onGround ? 1 : 0} R=${o.right ? 1 : 0} L=${o.left ? 1 : 0} J=${o.jump ? 1 : 0} ${JSON.stringify(eval(process.env.W ?? '0'))}`); };
  const r = laggedTrace(id, SOLUTIONS[id], +lag, 0, []);
  console.log(lines.slice(-50).join('\n'), JSON.stringify(r));
});
